Hardware control loop for auger drill, pH electrode servo, and dual-seed dispenser

// ============================================================================
// PROJECT: Ptestbot (Prototype v1.0)
// PURPOSE: Autonomous Soil Drilling, pH Sensing & Precision Seeding Rover
// ALIGNMENT: UN SDG 2 (Zero Hunger) & UN SDG 15 (Life on Land)
// ============================================================================

#include <ESP32Servo.h>

Servo drillActuator;   // Linear rack servo lowering the soil auger drill
Servo phProbeServo;    // Servo arm lowering analog pH electrode probe
Servo seedDispenser;   // Dual-chamber precision seed selector gate

const int PIN_DRILL_SERVO = 13;
const int PIN_PROBE_SERVO = 12;
const int PIN_SEED_SERVO  = 14;
const int PIN_DRILL_MOTOR = 27;
const int PIN_PH_ANALOG   = 34; // Analog pH sensor input (12-bit ADC: 0-4095)

void setup() {
  Serial.begin(115200);

  drillActuator.attach(PIN_DRILL_SERVO);
  phProbeServo.attach(PIN_PROBE_SERVO);
  seedDispenser.attach(PIN_SEED_SERVO);
  pinMode(PIN_DRILL_MOTOR, OUTPUT);

  // Stow all mechanical actuators safely before navigation
  drillActuator.write(0);
  phProbeServo.write(0);
  seedDispenser.write(90); // 90 deg = closed neutral gate

  Serial.println("[BOOT] Ptestbot Prototype v1.0 Firmware Initialized.");
}

void loop() {
  // 1. NAVIGATING: Move rover to next GPS grid coordinate
  Serial.println("\n[NAVIGATING] Moving to next field sector...");
  delay(1200);

  // 2. DRILLING: Bore 10 cm sample hole into topsoil
  Serial.println("[DRILLING] Engaging auger motor and lowering drill...");
  digitalWrite(PIN_DRILL_MOTOR, HIGH);
  drillActuator.write(90);
  delay(1800);
  drillActuator.write(0);
  digitalWrite(PIN_DRILL_MOTOR, LOW);
  delay(400);

  // 3. PROBING_PH: Insert pH sensor probe and read soil acidity
  Serial.println("[PROBING_PH] Lowering pH electrode into sample bore...");
  phProbeServo.write(90);
  delay(1500);

  int rawADC = analogRead(PIN_PH_ANALOG);
  float soilPH = 4.2 + (rawADC / 4095.0) * 4.1; // Maps ADC reading to pH 4.2 - 8.3
  Serial.print("[TELEMETRY] Measured Soil pH: ");
  Serial.println(soilPH, 1);

  phProbeServo.write(0); // Retract probe
  delay(400);

  // 4. EVALUATING & ACTUATING: Precision Crop & Soil Treatment Decision
  if (soilPH >= 5.0 && soilPH <= 5.9) {
    Serial.println("[PLANTING] Acidic Optimal -> Planting Crop A (Potatoes / Blueberries)");
    seedDispenser.write(35);  // Open Chamber A
    delay(500);
    seedDispenser.write(90);  // Close gate
  }
  else if (soilPH >= 6.0 && soilPH <= 7.2) {
    Serial.println("[PLANTING] Neutral Optimal -> Planting Crop B (Corn / Tomatoes)");
    seedDispenser.write(145); // Open Chamber B
    delay(500);
    seedDispenser.write(90);  // Close gate
  }
  else if (soilPH < 5.0) {
    Serial.println("[FLAGGING] Too Acidic (<5.0) -> Flagged for Agricultural Lime. Seed skipped.");
  }
  else {
    Serial.println("[FLAGGING] Too Alkaline (>7.2) -> Flagged for Sulfur Treatment. Seed skipped.");
  }

  delay(2000);
}
